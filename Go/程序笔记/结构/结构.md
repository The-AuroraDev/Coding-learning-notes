# 声明结构
文件位置：Playground/结构/struct-literal.go
```go
var person struct {   // 使用 var
	name string
	age int
}

person.name = "anan"
person.age = 13

fmt.Println(person.name)
fmt.Println(person.age)
```

# 由结构组成的切片
文件位置：Playground/结构/slice-struct.go
```go
type person struct {
		name string
		age int
	}

	persons := []person {
		{name: "anan", age: 13},
		{name: "jack", age: 14},
		{name: "tom", age: 10},
	}

	fmt.Println(persons)   // [{anan 13} {jack 14} {tom 10}]
	fmt.Println(persons[1].name)  // jack
```

# 将结构编码为 JSON
参考：[[Json 格式化输出]]
文件位置：Playground/结构/json.go， Playground/结构/json.json

## 实例代码
```go
type Address struct {
	City string `json:"city"`
}

type User struct {
	ID       int      `json:"id"`
	Name     string   `json:"name"`
	Email    string   `json:"email,omitempty"`
	Password string   `json:"-"`
	Tags     []string `json:"tags"`
	Address  Address  `json:"address"`
	active   bool     // 未导出，被忽略
}

func main() {
	u := User{
		ID:       1,
		Name:     "AnAn",
		Password: "secret",
		Tags:     []string{"dev", "go"},
		Address:  Address{City: "ChengDu"},
	}

	b, err := json.MarshalIndent(u, "", "  ")    // func MarshalIndent(v any, prefix, indent string) ([]byte, error)
	if err != nil {
		panic(err)
	}
	fmt.Println(string(b))
```

## 输出
```json
{
  "id": 1,
  "name": "AnAn",
  "tags": [
    "dev",
    "go"
  ],
  "address": {
    "city": "ChengDu"
  }
}

```
